# Folder Operations

Create, retrieve, and manage folders in the Eagle library.

## Import

```tsx
import { eagleApi } from '@/lib/eagle-api';
```

## Get All Folders

Retrieve all folders in the library.

```tsx
const folders = await eagleApi.folder.getAll();
```

**Returns:** `Promise<EagleFolder[]>`

### EagleFolder Type

```tsx
interface EagleFolder {
  id: string;              // Unique folder ID
  name: string;            // Folder name
  description: string;     // Folder description
  icon: string;            // Icon identifier
  iconColor: string;       // Icon color (hex)
  children: EagleFolder[]; // Nested subfolders
  modificationTime: number; // Last modified (Unix ms)
  tags: string[];          // Associated tags
  password: string;        // Folder password (if protected)
  passwordTips: string;    // Password hint
}
```

### Example: Build Folder Tree

```tsx
function FolderTree() {
  const [folders, setFolders] = useState<EagleFolder[]>([]);

  useEffect(() => {
    eagleApi.folder.getAll().then(setFolders);
  }, []);

  const renderFolder = (folder: EagleFolder, depth = 0) => (
    <div key={folder.id} style={{ marginLeft: depth * 16 }}>
      <span style={{ color: folder.iconColor }}>{folder.icon}</span>
      {folder.name}
      {folder.children.map(child => renderFolder(child, depth + 1))}
    </div>
  );

  return <div>{folders.map(f => renderFolder(f))}</div>;
}
```

## Get Selected Folder

Get the currently selected folder in Eagle's sidebar.

```tsx
const folder = await eagleApi.folder.getSelected();
```

**Returns:** `Promise<EagleFolder | null>`

```tsx
async function addToCurrentFolder(filePath: string) {
  const selectedFolder = await eagleApi.folder.getSelected();
  
  await eagleApi.item.addFromPath(filePath, {
    folders: selectedFolder ? [selectedFolder.id] : [],
  });
}
```

## Get Folder by ID

Retrieve a specific folder by its ID.

```tsx
const folder = await eagleApi.folder.getById(folderId);
```

**Returns:** `Promise<EagleFolder | null>`

```tsx
const folder = await eagleApi.folder.getById('FOLDER123');
if (folder) {
  console.log(`Folder: ${folder.name}`);
  console.log(`Subfolders: ${folder.children.length}`);
}
```

## Get Recent Folders

Get recently accessed folders.

```tsx
const recentFolders = await eagleApi.folder.getRecent();
```

**Returns:** `Promise<EagleFolder[]>`

```tsx
function RecentFolders() {
  const [folders, setFolders] = useState<EagleFolder[]>([]);

  useEffect(() => {
    eagleApi.folder.getRecent().then(setFolders);
  }, []);

  return (
    <div>
      <h3>Recent Folders</h3>
      {folders.map(folder => (
        <div key={folder.id}>{folder.name}</div>
      ))}
    </div>
  );
}
```

## Create Folder

Create a new folder in the library.

```tsx
const folder = await eagleApi.folder.create(options);
```

**Parameters:**

| Option | Type | Required | Description |
|--------|------|----------|-------------|
| `name` | `string` | Yes | Folder name |
| `parent` | `string` | No | Parent folder ID |
| `description` | `string` | No | Folder description |
| `icon` | `string` | No | Icon identifier |
| `iconColor` | `string` | No | Icon color (hex) |

**Returns:** `Promise<EagleFolder>`

### Example: Create Output Folder

```tsx
async function createOutputFolder() {
  const folder = await eagleApi.folder.create({
    name: `Plugin Output - ${new Date().toLocaleDateString()}`,
    description: 'Generated by My Plugin',
    icon: 'folder',
    iconColor: '#22c55e', // Green
  });
  
  return folder.id;
}
```

### Example: Create Nested Folder

```tsx
async function createSubfolder(parentId: string, name: string) {
  const folder = await eagleApi.folder.create({
    name,
    parent: parentId,
  });
  
  return folder;
}
```

## Update Folder

Modify folder properties.

```tsx
await eagleApi.folder.update(folderId, updates);
```

**Parameters:**

| Parameter | Type | Description |
|-----------|------|-------------|
| `folderId` | `string` | Folder ID to update |
| `updates.name` | `string` | New name |
| `updates.description` | `string` | New description |
| `updates.icon` | `string` | New icon |
| `updates.iconColor` | `string` | New color |

```tsx
await eagleApi.folder.update('FOLDER123', {
  name: 'Renamed Folder',
  iconColor: '#3b82f6', // Blue
});
```

## Navigate to Folder

Open a folder in Eagle's main window.

```tsx
await eagleApi.folder.open(folderId);
```

```tsx
// Navigate to folder after creating it
const folder = await eagleApi.folder.create({ name: 'New Folder' });
await eagleApi.folder.open(folder.id);
```

## Folder Utilities

### Flatten Folder Tree

```tsx
function flattenFolders(folders: EagleFolder[]): EagleFolder[] {
  return folders.reduce((acc, folder) => {
    return [...acc, folder, ...flattenFolders(folder.children)];
  }, [] as EagleFolder[]);
}

// Usage
const allFolders = await eagleApi.folder.getAll();
const flatList = flattenFolders(allFolders);
console.log(`Total folders: ${flatList.length}`);
```

### Find Folder by Name

```tsx
function findFolderByName(
  folders: EagleFolder[],
  name: string
): EagleFolder | undefined {
  for (const folder of folders) {
    if (folder.name === name) return folder;
    const found = findFolderByName(folder.children, name);
    if (found) return found;
  }
  return undefined;
}

// Usage
const allFolders = await eagleApi.folder.getAll();
const outputFolder = findFolderByName(allFolders, 'Plugin Output');
```

### Get Folder Path

```tsx
function getFolderPath(
  folders: EagleFolder[],
  targetId: string,
  path: string[] = []
): string[] | null {
  for (const folder of folders) {
    const currentPath = [...path, folder.name];
    if (folder.id === targetId) return currentPath;
    const found = getFolderPath(folder.children, targetId, currentPath);
    if (found) return found;
  }
  return null;
}

// Usage
const allFolders = await eagleApi.folder.getAll();
const path = getFolderPath(allFolders, 'FOLDER123');
// ['Root Folder', 'Subfolder', 'Target Folder']
```

## Browser vs Eagle Behavior

| Method | Browser (Mock) | Eagle |
|--------|----------------|-------|
| `getAll()` | Returns mock folders | Returns real folders |
| `getSelected()` | Returns null or mock | Returns selected folder |
| `create()` | Simulates creation | Creates real folder |
| `open()` | No-op | Navigates to folder |

## Best Practices

### ✅ Do

```tsx
// Cache folder list if needed multiple times
const folders = await eagleApi.folder.getAll();
const folderMap = new Map(flattenFolders(folders).map(f => [f.id, f]));

// Create descriptive folder names
await eagleApi.folder.create({
  name: `Export ${format.toUpperCase()} - ${date}`,
  description: 'Auto-generated by plugin',
});
```

### ❌ Don't

```tsx
// Don't call getAll() repeatedly in a loop
for (const item of items) {
  const folders = await eagleApi.folder.getAll(); // Expensive!
  // ...
}

// Don't create folders with duplicate names without checking
await eagleApi.folder.create({ name: 'Output' }); // May conflict
```
